cmake_minimum_required(VERSION 3.24)
project(BotWWindowsPortableLauncher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MINGW)
  message(FATAL_ERROR "Use a 64-bit LLVM-MinGW toolchain on Windows")
endif()
find_path(BOOST_INCLUDE_DIR boost/json.hpp HINTS "${BOOST_ROOT}" REQUIRED)
add_executable(BotWLauncher WIN32 launcher.cpp launcher_update_policy.cpp launcher_update_win.cpp)
target_include_directories(BotWLauncher PRIVATE "${BOOST_INCLUDE_DIR}")
target_compile_options(BotWLauncher PRIVATE -municode -O2)
target_link_options(BotWLauncher PRIVATE -municode -static)
target_link_libraries(BotWLauncher PRIVATE comctl32 dwmapi shell32 ole32 uuid dxgi winhttp bcrypt winpthread)
add_executable(BotWPlay WIN32 play_shortcut.cpp)
target_compile_options(BotWPlay PRIVATE -municode -O2)
target_link_options(BotWPlay PRIVATE -municode -static)
target_link_libraries(BotWPlay PRIVATE shell32 user32)
set_target_properties(BotWPlay PROPERTIES OUTPUT_NAME "Play Breath of the Wild")
include(CTest)
if(BUILD_TESTING)
  add_executable(launcher_settings_tests launcher_settings_tests.cpp)
  target_link_options(launcher_settings_tests PRIVATE -static)
  add_test(NAME settings COMMAND launcher_settings_tests)
  add_executable(launcher_update_tests launcher_update_tests.cpp launcher_update_policy.cpp)
  target_include_directories(launcher_update_tests PRIVATE "${BOOST_INCLUDE_DIR}")
  target_link_options(launcher_update_tests PRIVATE -static)
  add_test(NAME updates COMMAND launcher_update_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME portable_update_safety COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/portable_install_tests.py")
  endif()
endif()
